# --- Employees 360 Wave 3/3C: employment lifecycle (flag-gated, WATHEFNI allowlist) ---


class DashboardEmployeeLifecycleRequest(BaseModel):
    case_type: Literal[
        "termination",
        "cancel_scheduled",
        "reinstate",
        "rehire",
        "notice",
        "suspension",
        "unsuspend",
        "activate_start",
    ]
    reason: str = Field(min_length=3, max_length=500)
    approval_reference: str = Field(min_length=1, max_length=200)
    designated_approver_user_id: str
    idempotency_key: str = Field(min_length=8, max_length=200)
    expected_lifecycle_state: str
    expected_lifecycle_version: int
    expected_updated_at: datetime
    payload: dict[str, Any] | None = None
    impact_ack: bool = False
    impact_ack_text: str | None = Field(default=None, max_length=500)


class DashboardEmployeeLifecycleDecision(BaseModel):
    action: Literal["approve", "reject"]
    decision_reason: str | None = Field(default=None, max_length=500)


class DashboardEmployeeLifecyclePolicyPatch(BaseModel):
    tier: Literal["small", "medium", "enterprise"] | None = None
    timezone: str | None = None
    notice_hint_monthly_days: int | None = None
    notice_hint_other_days: int | None = None
    revoke_mode: Literal["end_of_last_working_day", "start_of_effective_date", "immediate_on_summary"] | None = None
    downstream_mode: Literal["warn_first", "block_submitted_payroll", "enterprise_configurable"] | None = None
    require_impact_ack: bool | None = None
    require_counsel_gate: bool | None = None
    rehire_same_employee_key: bool | None = None
    scheduler_cadence: str | None = None
    lag_alert_seconds: int | None = None


class DashboardEmployeeLifecycleCounselSign(BaseModel):
    answers: dict[str, Any]
    signed_by_name: str = Field(min_length=2, max_length=200)
    signed_by_role: str = Field(min_length=2, max_length=120)
    signed_reference: str = Field(min_length=2, max_length=200)
    notes: str | None = Field(default=None, max_length=1000)


class DashboardEmployeeLifecycleDownstreamRequest(BaseModel):
    action_type: Literal[
        "cancel_future_shifts",
        "decline_open_leave",
        "resolve_attendance_hold",
        "abandon_onboarding",
        "revoke_app_access",
        "restore_app_access",
        "reverse_cancel_future_shifts",
        "reverse_decline_open_leave",
        "reverse_abandon_onboarding",
    ]
    reason: str = Field(min_length=3, max_length=500)
    designated_approver_user_id: str
    idempotency_key: str = Field(min_length=8, max_length=200)
    payload: dict[str, Any] | None = None
    impact_ack: bool = False
    impact_ack_text: str | None = Field(default=None, max_length=500)

    import employee_lifecycle_wave3 as _w3

@app.get("/dashboard/posthire/employees/{employee_key}/lifecycle")
def dashboard_employee_lifecycle_get(
    employee_key: str,
    context: dict[str, Any] = Depends(dashboard_context),
):
    company = require_employee_roster_admin(context, "employees.manage")
    require_employee_mutation_scope(context, employee_key, company_code=company, action="lifecycle_read")
    import employee_lifecycle_wave3 as _w3
    import employee_lifecycle_wave3c as _lifecycle

    if not _w3.lifecycle_v3_enabled(company):
        raise HTTPException(status_code=403, detail={"error": "lifecycle_disabled", "message": "Employment lifecycle is not enabled for this company."})
    proj = _lifecycle.get_lifecycle_projection(sys.modules[__name__], company_code=company, employee_key=employee_key)
    if not proj:
        raise HTTPException(status_code=404, detail={"error": "employee_not_found", "message": "We couldn't find that employee."})
    return json_safe({"ok": True, "lifecycle": proj})

@app.get("/dashboard/posthire/employee-lifecycle/pending")
def dashboard_employee_lifecycle_pending(
    employee_key: str | None = None,
    context: dict[str, Any] = Depends(dashboard_context),
):
    company = context["company_code"]
    if not any(company_has_module(company, m) for m in POSTHIRE_PEOPLE_MODULES):
        raise HTTPException(status_code=403, detail={"error": "module_disabled", "message": "This module is not enabled for your company."})
    if not (
        dashboard_context_has_permission(context, "employees.manage")
        or dashboard_context_has_permission(context, "employees.status.approve")
    ):
        raise HTTPException(status_code=403, detail={"error": "permission_denied", "message": "You do not have access to do that."})
    import employee_lifecycle_wave3 as _w3
    import employee_lifecycle_wave3c as _lifecycle

    if not _w3.lifecycle_v3_enabled(company):
        raise HTTPException(status_code=403, detail={"error": "lifecycle_disabled"})
    rows = _lifecycle.list_pending_lifecycle_for_actor(
        sys.modules[__name__],
        company_code=company,
        actor_user_id=str(context.get("actor_user_id") or ""),
        employee_key=employee_key,
    )
    return {"ok": True, "company_code": company, "count": len(rows), "requests": rows}
